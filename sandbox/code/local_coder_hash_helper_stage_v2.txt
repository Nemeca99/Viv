import hashlib

def hash_file(path):
    """Return SHA-256 hex digest of a file."""
    sha_signature = hashlib.sha256()
    with open(path, "rb") as f:
        for byte_block in iter(lambda: f.read(4096), b""):
            sha_signature.update(byte_block)
    return sha_signature.hexdigest()
