        + str(ev.get("align_verdict"))
        + " exec_v2="
        + str(ev.get("viv_executes_v2"))
    )

                "master_s_n": ev.get("master_s_n"),
                "master_status": ev.get("master_status"),
                "align_verdict": ev.get("align_verdict"),
