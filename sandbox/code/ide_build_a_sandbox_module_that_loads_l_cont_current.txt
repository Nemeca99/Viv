from pathlib import Path
import json
p = Path('L:/Continue/Viv/sandbox/work/aios_build/system_bridges/rag_core_bridge.json')
if not p.is_file():
    print('JSON missing', p)
else:
    data = json.loads(p.read_text(encoding='utf-8'))
    if isinstance(data, dict):
        keys = ['system_id','role','on_disk','status','file_sample','viv_map']
        for k in keys:
            if k in data:
                print(k + '=' + str(data.get(k))[:200])
        print('keys=' + str(list(data.keys())[:20]))
    else:
        print(type(data), str(data)[:300])
