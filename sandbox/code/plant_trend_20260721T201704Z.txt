from pathlib import Path
import re
p = Path(r"L:/Continue/Viv/sandbox/work/plant_brief.txt")
if not p.is_file():
    print("PLANT_TREND missing")
else:
    sns = []
    for line in p.read_text(encoding="utf-8").splitlines():
        m = re.search(r"S_n=([0-9.]+)", line)
        if m:
            sns.append(float(m.group(1)))
    if not sns:
        print("PLANT_TREND empty")
    else:
        print(f"PLANT_TREND n={len(sns)} last={sns[-1]:.4f} "
              f"min={min(sns):.4f} max={max(sns):.4f} "
              f"delta={sns[-1]-sns[0]:+.4f}")
